=== POU Main (Program) ===
(* Run, dropped by a jam. *)
Conveyor := (Start_PB OR Conveyor) AND Stop_PB AND EStop_OK AND NOT Jam_Alarm;

(* Product blocking the discharge photocell while running. *)
Jam_Timer(IN := PE_Discharge AND Conveyor, PT := 5000);

(* Latched, so it needs a deliberate reset. *)
IF Jam_Timer.DN THEN
  Jam_Alarm := TRUE;
END_IF;
